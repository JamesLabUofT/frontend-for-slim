viewof copyButton = {
  const btn = html`<button class="btn btn-outline-primary">📋 Copy Command</button>`;
  
  btn.onclick = () => {
    navigator.clipboard.writeText(command)
      .then(() => {
        // Temporary feedback on successful copy
        const originalText = btn.innerHTML;
        btn.innerHTML = "✅ Copied!";
        setTimeout(() => btn.innerHTML = originalText, 1500);
      })
      .catch(err => console.error("Clipboard copy failed:", err));
  };
  
  return btn;
}
rawFileURL = "https://raw.githubusercontent.com/JamesLabUofT/frontend-for-slim/refs/heads/main/scripts/phase1_slim_script.slim"
response = await fetch(rawFileURL); 
blob = await response.blob();
viewof downloadSlimScript = {
  const downloadButton = html`<button class="btn btn-outline-primary" download = "slimFile">Download SLiM file</button>`;
  downloadButton.onclick = () => {
    
    try {
      
      if (!response.ok) {
        throw new Error(`HTTP error while fetching slim script! Status : ${response.status}`);
      }
      
      const blobUrl = URL.createObjectURL(blob);
      
      const a = document.createElement('a');
      a.href = blobUrl;
      a.download = "slimScript.slim";
      a.style.display = 'none';
      
      a.click();
      
      URL.revokeObjectURL(blobUrl);
      
      
    } catch (error) {
      console.error('Failed to download file:', error);
      alert('Could not download the file.')
    }
  }
  return downloadButton;
}
